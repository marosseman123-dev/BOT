# Unified bot deps. Reuses ./seo_audit and ./EmailHunterV2/hunter in place.
# httpx (pure-Python) is used instead of aiohttp: aiohttp's C extensions crash
# with an access violation on CPython 3.14. httpx runs 143k+ domains stably there.
httpx>=0.27            # async HTTP client (single-crawl fetch)
brotli>=1.1            # decode 'br'-encoded pages (Cloudflare/nginx) instead of failing
# httpx no longer auto-detects charset even when this is installed -- it has to be
# wired in explicitly (see fetch.py's _get()). Without it, any page that doesn't
# declare its charset in the Content-Type header gets force-decoded as UTF-8;
# older/smaller SK/CZ sites (cp1250/iso-8859-2, very common on this exact target
# list) turn every diacritic byte into a replacement character, corrupting
# emails and text (confirmed on a live sample: "objednavky" with the accented
# letter mangled into junk instead of decoding correctly).
charset-normalizer>=3.3
beautifulsoup4>=4.12   # HTML parsing (seo_audit + classifier)
lxml>=5.0              # fast parser backend
pandas>=2.0            # CSV chunked read
certifi>=2024.1.1
